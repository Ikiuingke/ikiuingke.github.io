<!-- https://github.com/isomorphic-git/lightning-fs -->
<!-- https://isomorphic-git.org/docs/en/quickstart -->
<script src="https://unpkg.com/@isomorphic-git/lightning-fs"></script>
<script src="https://unpkg.com/isomorphic-git@beta"></script>
<script type="module">
import http from 'https://unpkg.com/isomorphic-git@beta/http/web/index.js'

window.fs = new LightningFS('fs')
window.pfs = window.fs.promises

const cwd = "/data"
const sitesDir = `${cwd}/content/sites`

const dirData = await pfs.readdir(cwd);

if (!dirData) {
  await pfs.mkdir(cwd);
	await git.clone({
		fs,
		http,
		dir: cwd,
		corsProxy: 'https://cors.isomorphic-git.org',
		url: 'https://gitlab.com/vinyl-finder/data.git',
		ref: 'main',
		singleBranch: true,
		depth: 10
	});
}
const siteFiles = await pfs
	.readdir(sitesDir)

const sitePromises = await siteFiles
	.filter(file => file.endsWith(".json"))
	.map((siteFile) => {
		return pfs
			.readFile(`${sitesDir}/${siteFile}`, {encoding: "utf8"})
			.then(JSON.parse)
  })

const sites = await Promise.all(sitePromises)

const vfSites = document.querySelector('vf-sites')
vfSites.setAttribute("sites", JSON.stringify(sites))
</script>
#+end_export
